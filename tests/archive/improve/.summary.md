## API Contract

### applyImprovements(improvements, options)
```javascript
// Input
const improvements = [
  {
    improvementId: 'fix-english-switching',
    file: '/absolute/path/to/file.js',
    oldString: 'old code',
    newString: 'new code',
    description: 'Fix description',
    category: 'bug-fix'
  }
];

const options = {
  auto: false,           // Auto-apply without confirmation
  iteration: null,       // Test iteration number
  createBackup: true,    // Create backups
  commit: true,          // Create git commit
  tag: true              // Create git tag
};

// Output
{
  applied: [
    {
      improvementId: 'fix-english-switching',
      status: 'applied',
      file: '/path/to/file.js',
      backupPath: '/path/to/.backups/file.js.TIMESTAMP.backup'
    }
  ],
  skipped: [
    {
      improvementId: 'other-fix',
      reason: 'user declined',
      file: '/path/to/other.js'
    }
  ],
  commit: 'abc123def456',
  tag: 'test-iteration-5'
}
```

### showDiff(improvement)
```javascript
// Input
const improvement = { improvementId, file, oldString, newString, description };

// Output
{
  filepath: '/absolute/path/to/file.js',
  diff: '--- a/file.js\n+++ b/file.js\n@@ -1,1 +1,1 @@\n-old\n+new',
  willApply: true,
  newContent: 'full file content after change'
}
```

### rollbackIteration(commitHash)
```javascript
// Rollback to a specific commit or tag
await rollbackIteration('test-iteration-4');
await rollbackIteration('abc123def456');
await rollbackIteration('HEAD~1');
```

## Usage Examples

### CLI Usage
```bash
# Manual mode with confirmation
node apply-improvements.mjs --input improvements.json

# Auto-apply with iteration tracking
node apply-improvements.mjs --input improvements.json --auto --iteration 5

# Rollback to previous iteration
node apply-improvements.mjs --rollback test-iteration-4
```

### Programmatic Usage
```javascript
import { applyImprovements, showDiff, rollbackIteration } from './apply-improvements.mjs';

// Apply improvements
const result = await applyImprovements(improvements, { auto: true, iteration: 5 });

// Show diff only
showDiff(improvement);

// Rollback if needed
await rollbackIteration('test-iteration-4');
```
